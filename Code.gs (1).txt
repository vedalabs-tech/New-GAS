function doGet(e) {
  try {
    if (!e || !e.parameter || e.parameter.token !== APP_TOKEN) {
      return jsonResponse({status: "error", message: "Access Denied! Invalid or missing security token."});
    }
    const action = e.parameter.action;

    if (action === "getProducts") return jsonResponse(getProductsFromSheet(e.parameter.includeScheduled === "true"));
    if (action === "getCategories") return jsonResponse(getCategoriesTree(e.parameter.includeHidden === "true"));
    if (action === "getOrderHistory") return jsonResponse(getOrderHistory(e.parameter.email));
    if (action === "trackOrder") return jsonResponse(trackOrder(e.parameter.query || e.parameter.orderId || e.parameter.email));
    if (action === "getCart") return jsonResponse(getCart(e.parameter.email));
    if (action === "getReviews") return jsonResponse(getApprovedReviews(e.parameter.productId));
    if (action === "getAllReviewsAdmin") return jsonResponse(getAllReviewsForAdmin());
    if (action === "getBanners") return jsonResponse(getBannersFromSheet(e.parameter.includeInactive === "true"));
    if (action === "getUserTypes") return jsonResponse(getUserTypesFromSheet(e.parameter.includeInactive === "true"));
    if (action === "getStoreConfig") {
      const liveKey = PropertiesService.getScriptProperties().getProperty("RAZORPAY_KEY_ID") || RAZORPAY_LIVE_KEY;
      return jsonResponse({status: "success", rzpKey: liveKey});
    }
    if (action === "getAddresses") return jsonResponse(getAddressesBook(e.parameter.email));
    if (action === "readUniversal") return jsonResponse(getUniversalData(e.parameter.sheetName));

    return jsonResponse({status: "error", message: "Invalid GET Action: " + (action || "none")});
  } catch (err) {
    return jsonResponse({status: "error", message: "Server error during doGet: " + err.message});
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({status: "error", message: "No data payload received."});
    }

    let data;
    try {
      data = JSON.parse(e.postData.contents);
    } catch(parseErr) {
      return jsonResponse({status: "error", message: "Malformed JSON payload: " + parseErr.message});
    }

    if (!data || data.token !== APP_TOKEN) {
      return jsonResponse({status: "error", message: "Access Denied! Invalid security token."});
    }
    const action = data.action;

    // AUTH ACTIONS (With UserType Personalization)
    if (action === "requestRegisterOTP") return jsonResponse(requestRegisterOTP(data.name, data.email, data.phone, data.userType));
    if (action === "verifyRegisterOTP") return jsonResponse(verifyRegisterOTP(data.email, data.otp, data.deviceInfo));
    if (action === "requestLoginOTP") return jsonResponse(requestLoginOTP(data.email));
    if (action === "verifyLoginOTP") return jsonResponse(verifyLoginOTP(data.email, data.otp, data.deviceInfo));

    // CUSTOMER ACTIONS
    if (action === "updateProfile") return jsonResponse(updateSecureProfile(data));
    if (action === "saveAddresses") return jsonResponse(saveAddressesBook(data.email, data.addresses || data.Addresses_JSON));
    if (action === "placeOrder") return jsonResponse(processOrder(data));
    if (action === "syncCart") return jsonResponse(syncCart(data.email, data.cartItems));
    if (action === "addWaitlist") return jsonResponse(addToWaitlist(data.email, data.productId));
    if (action === "addReviewWithMedia") return jsonResponse(addReviewWithMedia(data));
    if (action === "logActivity") return jsonResponse(logActivity(data.email, data.activityType, data.details)); 

    // ADVANCED CATEGORY MANAGEMENT (Admin)
    if (action === "saveCategory") return jsonResponse(saveCategory(data.category));
    if (action === "deleteCategory") return jsonResponse(deleteCategory(data.categoryId || data.categoryName));
    if (action === "toggleCategoryStatus") return jsonResponse(toggleCategoryStatus(data.categoryId, data.status));

    // PRODUCT MANAGEMENT (Admin with Unlimited Media, COD Toggle, Auto-Stock Scheduling)
    if (action === "saveProduct") return jsonResponse(saveProduct(data.product));
    if (action === "deleteProduct") return jsonResponse(deleteProduct(data.productId));

    // REVIEW MODERATION (Admin)
    if (action === "moderateReview") return jsonResponse(moderateReview(data.reviewId, data.status));

    // DYNAMIC BANNER MANAGEMENT (Admin & Storefront Dynamic Connection)
    if (action === "saveBanner") return jsonResponse(saveBanner(data.banner));
    if (action === "deleteBanner") return jsonResponse(deleteBanner(data.bannerId));
    if (action === "toggleBannerStatus") return jsonResponse(toggleBannerStatus(data.bannerId, data.status));
    if (action === "reorderBanners") return jsonResponse(reorderBanners(data.bannerIds));

    // DYNAMIC USER TYPE (PERSON TYPE) MANAGEMENT (Admin)
    if (action === "saveUserType") return jsonResponse(saveUserType(data.userType));
    if (action === "deleteUserType") return jsonResponse(deleteUserType(data.userTypeId));
    if (action === "toggleUserTypeStatus") return jsonResponse(toggleUserTypeStatus(data.userTypeId, data.status));

    // MEDIA UPLOADS (Admin Product Media & Customer Review Media)
    if (action === "uploadMedia") return jsonResponse(uploadMediaWithFallback(data.base64, data.filename, data.mimeType));

    // ADMIN OPERATIONS
    if (action === "trackOrder") return jsonResponse(trackOrder(data.query || data.orderId || data.email));
    if (action === "updateOrderStatus") return jsonResponse(updateOrderStatusAndAlert(data.orderId, data.status, data.courierName, data.trackingNumber, data.estimatedDelivery));
    if (action === "universalWrite") return jsonResponse(universalWrite(data.sheetName, data.rowData)); 
    if (action === "universalWriteDynamic") return jsonResponse(universalWriteDynamic(data.sheetName, data.payload)); 
    if (action === "universalUpdate") return jsonResponse(universalUpdate(data.sheetName, data.searchCol, data.searchValue, data.updateCol, data.updateValue));
    if (action === "sendAdminReply") return jsonResponse(sendAdminReply(data.email, data.replyText));
    if (action === "sendBulkPromo") return jsonResponse(sendBulkPromo(data.title, data.message, data.bannerUrl));

    return jsonResponse({status: "error", message: "Invalid POST Action: " + (action || "none")});
  } catch (err) {
    return jsonResponse({status: "error", message: "Server error during doPost: " + err.message});
  }
}

function jsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateFolder(parentFolder, folderName) {
  if (!parentFolder) parentFolder = DriveApp.getRootFolder();
  try {
    const folders = parentFolder.getFoldersByName(folderName);
    return folders.hasNext() ? folders.next() : parentFolder.createFolder(folderName);
  } catch(e) {
    console.warn("Folder access warning:", e.message);
    return DriveApp.getRootFolder();
  }
}

function getMasterDB() {
  const id = PropertiesService.getScriptProperties().getProperty("MASTER_SHEET_ID");
  if (id) {
    try {
      return SpreadsheetApp.openById(id);
    } catch(e) {
      console.warn("Could not open spreadsheet by MASTER_SHEET_ID:", e.message);
    }
  }

  // Fallback to active spreadsheet if script is container-bound
  try {
    const active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch(e) {}

  // Fallback to auto-setting up ecosystem
  const setupResult = setupEcosystem();
  if (setupResult && setupResult.masterSheetId) {
    return SpreadsheetApp.openById(setupResult.masterSheetId);
  }

  throw new Error("System is not ready. MASTER_DB could not be located. Please run setupEcosystem().");
}

function ensureDynamicColumns(sheet, dataObject) {
  if (!sheet) return [];
  let lastCol = sheet.getLastColumn();

  if (lastCol === 0) {
    let initialHeaders = Object.keys(dataObject).filter(k => k !== "action" && k !== "token" && k !== "email");
    if (initialHeaders.length > 0) {
      sheet.appendRow(initialHeaders);
      SpreadsheetApp.flush();
      return initialHeaders;
    }
    return [];
  }

  let headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  let lowerHeaders = headers.map(h => String(h).toLowerCase().trim());
  let columnsAdded = false;

  Object.keys(dataObject).forEach(key => {
    if (key !== "action" && key !== "token" && key !== "email") {
      let lowerKey = String(key).toLowerCase().trim();
      if (!lowerHeaders.includes(lowerKey)) {
        sheet.getRange(1, headers.length + 1).setValue(key);
        headers.push(key);
        lowerHeaders.push(lowerKey);
        columnsAdded = true;
      }
    }
  });

  if (columnsAdded) SpreadsheetApp.flush();
  return headers;
}

function getUniversalData(sheetName) {
  try {
    SpreadsheetApp.flush(); 
    const master = getMasterDB();
    const sheet = master.getSheetByName(sheetName);
    if (!sheet) return [];
    
    const lastRow = sheet.getLastRow();
    const lastCol = sheet.getLastColumn();
    if (lastRow <= 1 || lastCol === 0) return [];

    const data = sheet.getRange(1, 1, lastRow, lastCol).getValues();
    if (data.length <= 1) return [];
    
    const rawHeaders = data.shift(); 
    const headers = rawHeaders.map(h => String(h).trim());

    return data.map(row => { 
      let obj = {}; 
      headers.forEach((h, i) => {
        if (h) {
          let val = row[i];
          if (val instanceof Date) {
            val = val.toISOString();
          }
          obj[h] = val;
        }
      }); 

      // Safe address lookup for user rows
      if (sheetName === "Users") {
        let addrFileId = obj["AddressFileID"];
        if (addrFileId) {
          try {
            let addrSheet = SpreadsheetApp.openById(addrFileId).getSheetByName("Saved_Addresses");
            if (addrSheet && addrSheet.getLastRow() > 1) {
              let aData = addrSheet.getRange(1, 1, addrSheet.getLastRow(), 5).getValues();
              let aArr = [];
              for (let r = 1; r < aData.length; r++) {
                aArr.push({ 
                  l1: String(aData[r][0] || ""), 
                  l2: String(aData[r][1] || ""), 
                  city: String(aData[r][2] || ""), 
                  state: String(aData[r][3] || ""), 
                  pin: String(aData[r][4] || "") 
                });
              }
              obj["Addresses_JSON"] = JSON.stringify(aArr); 
            } else { 
              obj["Addresses_JSON"] = "[]"; 
            }
          } catch(e) { 
            obj["Addresses_JSON"] = "[]"; 
          }
        } else {
          obj["Addresses_JSON"] = "[]";
        }
      }
      return obj; 
    });
  } catch(e) {
    console.warn("Error reading universal data for sheet " + sheetName + ":", e.message);
    return [];
  }
}

function universalWriteDynamic(sheetName, payloadObj) {
  if (!payloadObj || typeof payloadObj !== "object") {
    return {status: "error", message: "Invalid payload object provided."};
  }

  const master = getMasterDB();
  let sheet = master.getSheetByName(sheetName);
  if (!sheet) {
    sheet = master.insertSheet(sheetName);
    sheet.appendRow(Object.keys(payloadObj));
  }
  
  let headers = ensureDynamicColumns(sheet, payloadObj);
  let newRow = new Array(headers.length).fill("");
  
  Object.keys(payloadObj).forEach(k => {
    let i = headers.indexOf(k);
    if (i > -1) {
      let val = payloadObj[k];
      if (typeof val === "object" && val !== null) val = JSON.stringify(val);
      newRow[i] = val;
    }
  });

  sheet.appendRow(newRow);
  SpreadsheetApp.flush();
  return {status: "success"};
}

function universalWrite(sheetName, rowDataArray) {
  const master = getMasterDB();
  let sheet = master.getSheetByName(sheetName);
  if (!sheet) sheet = master.insertSheet(sheetName); 

  let processedArray = Array.isArray(rowDataArray) ? rowDataArray.map(item => {
    if (typeof item === "object" && item !== null && !(item instanceof Date)) {
      return JSON.stringify(item);
    }
    return item;
  }) : [rowDataArray];

  sheet.appendRow(processedArray);
  SpreadsheetApp.flush();
  return {status: "success"};
}

function universalUpdate(sheetName, searchColIndex, searchValue, updateColIndex, updateValue) {
  const master = getMasterDB();
  const sheet = master.getSheetByName(sheetName);
  if (!sheet) return {status: "error", message: "Target sheet '" + sheetName + "' not found."};

  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= 1) return {status: "error", message: "No data rows exist in sheet."};

  if (searchColIndex < 0 || updateColIndex < 0 || searchColIndex >= lastCol) {
    return {status: "error", message: "Invalid column index specified."};
  }

  const data = sheet.getRange(1, 1, lastRow, lastCol).getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][searchColIndex]).trim() === String(searchValue).trim()) {
      let val = updateValue;
      if (typeof val === "object" && val !== null && !(val instanceof Date)) {
        val = JSON.stringify(val);
      }
      sheet.getRange(i + 1, updateColIndex + 1).setValue(val);
      SpreadsheetApp.flush();
      return {status: "success"};
    }
  }
  return {status: "error", message: "Record matching '" + searchValue + "' not found."};
}

function uploadMediaWithFallback(base64Data, filename, mimeType) {
  try {
    if (!base64Data) return {status: "error", message: "No media data provided."};
    
    // Clean data URL prefix if present
    const cleanBase64 = String(base64Data).replace(/^data:[^;]+;base64,/, '');
    const cleanMime = mimeType || (filename && filename.endsWith(".mp4") ? "video/mp4" : "image/jpeg");
    const isVideo = cleanMime.startsWith("video/") || (filename && /\.(mp4|webm|mov|mkv)$/i.test(filename));
    const cleanName = filename || (isVideo ? `video_${Date.now()}.mp4` : `photo_${Date.now()}.jpg`);

    const folderId = PropertiesService.getScriptProperties().getProperty("MEDIA_VAULT_ID");
    let folder;
    try {
      folder = folderId ? DriveApp.getFolderById(folderId) : DriveApp.getRootFolder();
    } catch(e) {
      folder = DriveApp.getRootFolder();
    }

    const blob = Utilities.newBlob(Utilities.base64Decode(cleanBase64), cleanMime, cleanName);
    const file = folder.createFile(blob);
    try {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch(e) {}
    
    const fileId = file.getId();
    const mediaType = isVideo ? "video" : "image";
    const directUrl = isVideo 
      ? `https://drive.google.com/file/d/${fileId}/preview` 
      : `https://drive.google.com/uc?export=view&id=${fileId}`;

    return {
      status: "success", 
      type: mediaType,
      fileId: fileId,
      url: directUrl,
      downloadUrl: `https://drive.google.com/uc?export=download&id=${fileId}`
    };
  } catch (error) {
    console.warn("Upload to Drive failed, using data URI fallback:", error.message);
    const safeMime = mimeType || "image/jpeg";
    const dataUri = base64Data.startsWith("data:") ? base64Data : ("data:" + safeMime + ";base64," + base64Data);
    return {
      status: "success", 
      type: safeMime.startsWith("video/") ? "video" : "image",
      url: dataUri, 
      fallback: true
    };
  }
}

// Backward compatibility alias
function uploadImageWithFallback(base64Data, filename, mimeType) {
  return uploadMediaWithFallback(base64Data, filename, mimeType);
}

  } catch(e) {
    console.warn("createSecurityLog failed gracefully:", e.message);
  }
}

function findRowIndexByEmailCaseInsensitive(email, sheet) {
  if (!email || !sheet) return -1;
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return -1;
  
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const emailColIdx = headers.findIndex(h => String(h).toLowerCase() === "email");
  if (emailColIdx === -1) return -1;

  const emailColValues = sheet.getRange(1, emailColIdx + 1, lastRow, 1).getValues();
  const searchEmail = String(email).toLowerCase().trim();

  for (let i = 1; i < emailColValues.length; i++) {
    if (String(emailColValues[i][0]).toLowerCase().trim() === searchEmail) {
      return i + 1; // 1-indexed row number in Google Sheets
    }
  }
  return -1;
}

function requestRegisterOTP(name, email, phone, userType) {
  if (!email || !String(email).includes("@")) {
    return {status: "error", message: "A valid email address is required."};
  }
  if (!name || String(name).trim().length < 2) {
    return {status: "error", message: "Full Name is required (minimum 2 characters)."};
  }

  const cleanName = String(name).trim();
  const cleanEmail = String(email).toLowerCase().trim();
  const cleanPhone = phone ? String(phone).trim() : "";
  const cleanUserType = userType ? String(userType).trim() : "Retail Customer";

  const sheet = getMasterDB().getSheetByName("Users");
  if (!sheet) {
    return {status: "error", message: "Users database sheet is not available. Please run setupEcosystem()."};
  }

  const existingRow = findRowIndexByEmailCaseInsensitive(cleanEmail, sheet);
  if (existingRow > 0) {
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const statusIdx = headers.findIndex(h => String(h).toLowerCase() === "status");
    const existingStatus = statusIdx > -1 ? sheet.getRange(existingRow, statusIdx + 1).getValue() : "";
    
    if (existingStatus === "Active") {
      return {status: "error", message: "An active account with this email already exists. Please log in directly."};
    }
  }

  const otp = Math.floor(100000 + Math.random() * 900000).toString();
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  
  if (existingRow > 0) {
    // Update existing pending record
    let nameIdx = headers.findIndex(h => String(h).toLowerCase() === "name");
    let phoneIdx = headers.findIndex(h => String(h).toLowerCase() === "phone");
    let typeIdx = headers.findIndex(h => String(h).toLowerCase() === "usertype");
    let otpIdx = headers.findIndex(h => String(h).toLowerCase() === "otp");
    
    if (nameIdx > -1) sheet.getRange(existingRow, nameIdx + 1).setValue(cleanName);
    if (phoneIdx > -1) sheet.getRange(existingRow, phoneIdx + 1).setValue(cleanPhone);
    if (typeIdx > -1) sheet.getRange(existingRow, typeIdx + 1).setValue(cleanUserType);
    if (otpIdx > -1) sheet.getRange(existingRow, otpIdx + 1).setValue(otp);
  } else {
    // Create new user row with UserType personalization
    let newRow = new Array(headers.length).fill("");
    headers.forEach((h, i) => {
      let lower = String(h).toLowerCase().trim();
      if (lower === "name") newRow[i] = cleanName;
      else if (lower === "email") newRow[i] = cleanEmail;
      else if (lower === "phone") newRow[i] = cleanPhone;
      else if (lower === "usertype") newRow[i] = cleanUserType;
      else if (lower === "otp") newRow[i] = otp;
      else if (lower === "status") newRow[i] = "Pending_OTP";
      else if (lower === "visits") newRow[i] = 0;
      else if (lower === "updatecount") newRow[i] = 0;
    });
    sheet.appendRow(newRow);
  }
  
  SpreadsheetApp.flush();
  sendOTPEmail(cleanEmail, otp, "Registration");
  return {status: "success", message: `Verification OTP has been dispatched to ${cleanEmail}.`};
}

function parseAddressList(raw) {
  if (!raw) return [];
  let list = raw;
  if (typeof raw === "string") {
    try { list = JSON.parse(raw); } catch (e) { return []; }
  }
  if (!Array.isArray(list)) return [];
  return list.map(function(row) {
    const rec = row || {};
    return {
      l1: String(rec.l1 || rec.Address_Line_1 || rec.line1 || ""),
      l2: String(rec.l2 || rec.Address_Line_2 || rec.line2 || ""),
      city: String(rec.city || rec.City || ""),
      state: String(rec.state || rec.State || ""),
      pin: String(rec.pin || rec.pincode || rec.Pincode || "")
    };
  }).filter(function(row) {
    return row.l1 || row.city || row.pin;
  });
}

function readAddressBookForEmail(email) {
  if (!email) return [];
  try {
    const sheet = getMasterDB().getSheetByName("Users");
    const rowIndex = findRowIndexByEmailCaseInsensitive(email, sheet);
    if (rowIndex < 1) return [];
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const fileIdIndex = headers.findIndex(function(h) { return String(h).toLowerCase() === "addressfileid"; });
    if (fileIdIndex < 0) return [];
    const fileId = sheet.getRange(rowIndex, fileIdIndex + 1).getValue();
    if (!fileId) return [];
    const addrSheet = SpreadsheetApp.openById(fileId).getSheetByName("Saved_Addresses");
    if (!addrSheet || addrSheet.getLastRow() <= 1) return [];
    const aData = addrSheet.getRange(1, 1, addrSheet.getLastRow(), 5).getValues();
    const aArr = [];
    for (let r = 1; r < aData.length; r++) {
      const row = {
        l1: String(aData[r][0] || ""),
        l2: String(aData[r][1] || ""),
        city: String(aData[r][2] || ""),
        state: String(aData[r][3] || ""),
        pin: String(aData[r][4] || "")
      };
      if (row.l1 || row.city || row.pin) aArr.push(row);
    }
    return aArr;
  } catch (e) {
    console.warn("readAddressBookForEmail:", e.message);
    return [];
  }
}

function updateSecureProfile(data) {
  if (!data || !data.email) return {status: "error", message: "Email is required."};
  const sheet = getMasterDB().getSheetByName("Users");
  const rowIndex = findRowIndexByEmailCaseInsensitive(data.email, sheet);

  if (rowIndex > 0) {
    // Ensure all critical tracking headers exist dynamically
    let headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    const requiredTrackingCols = ["UpdateCount", "Month", "UserID", "VaultID", "AddressFileID", "UserType", "ProfilePic"];
    let colsAdded = false;

    requiredTrackingCols.forEach(col => {
      let idx = headers.findIndex(h => String(h).toLowerCase() === col.toLowerCase());
      if (idx === -1) {
        sheet.getRange(1, headers.length + 1).setValue(col);
        headers.push(col);
        colsAdded = true;
      }
    });

    if (colsAdded) {
      SpreadsheetApp.flush();
      headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    }

    const currentMonth = new Date().getMonth() + 1;
    let limitCountIndex = headers.findIndex(h => String(h).toLowerCase() === "updatecount");
    let limitMonthIndex = headers.findIndex(h => String(h).toLowerCase() === "month");
    let nameIndex = headers.findIndex(h => String(h).toLowerCase() === "name");
    let phoneIndex = headers.findIndex(h => String(h).toLowerCase() === "phone");
    let picIndex = headers.findIndex(h => String(h).toLowerCase() === "profilepic");
    let typeIndex = headers.findIndex(h => String(h).toLowerCase() === "usertype");

    let recordedMonth = Number(sheet.getRange(rowIndex, limitMonthIndex + 1).getValue() || 0);
    let count = Number(sheet.getRange(rowIndex, limitCountIndex + 1).getValue() || 0);

    if (recordedMonth !== currentMonth) {
      count = 0;
      sheet.getRange(rowIndex, limitMonthIndex + 1).setValue(currentMonth);
    }
    
    // Safety limit: max 5 profile updates per month
    if (count >= 5) {
      return {status: "error", message: "Monthly profile update limit reached (maximum 5 updates per calendar month)."};
    }

    if (data.name && nameIndex > -1) sheet.getRange(rowIndex, nameIndex + 1).setValue(data.name);
    if (data.phone && phoneIndex > -1) sheet.getRange(rowIndex, phoneIndex + 1).setValue(data.phone);
    if (data.profilePic && picIndex > -1) sheet.getRange(rowIndex, picIndex + 1).setValue(data.profilePic);
    if (data.userType && typeIndex > -1) sheet.getRange(rowIndex, typeIndex + 1).setValue(data.userType);
    
    sheet.getRange(rowIndex, limitCountIndex + 1).setValue(count + 1);

    // Full address book from the storefront replaces the vault.
    // A single line still appends for older clients.
    const addressPayload = data.addresses || data.Addresses_JSON;
    if (addressPayload) {
      saveAddressesBook(data.email, addressPayload);
    } else if (data.addressLine1 || data.pincode) {
      saveDedicatedAddress(data.email, data.addressLine1, data.addressLine2, data.city, data.state, data.pincode);
    }

    SpreadsheetApp.flush();
    return {status: "success", message: "Profile successfully updated."};
  }
  return {status: "error", message: "User record not found."};
}

function getCategoriesTree(includeHidden) {
  const db = getMasterDB();
  const sheet = db.getSheetByName("Categories");
  if (!sheet || sheet.getLastRow() <= 1) return [];

  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(h => String(h).toLowerCase().trim());
  
  const idIdx = headers.findIndex(h => h === "categoryid");
  const nameIdx = headers.findIndex(h => h === "categoryname");
  const parentIdx = headers.findIndex(h => h === "parentcategoryid");
  const statusIdx = headers.findIndex(h => h === "status");
  const dateIdx = headers.findIndex(h => h === "scheduled_live_date");

  const now = new Date();
  const list = [];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const catId = String(row[idIdx > -1 ? idIdx : 0] || "").trim();
    const catName = String(row[nameIdx > -1 ? nameIdx : 1] || "").trim();
    const parentId = parentIdx > -1 ? String(row[parentIdx] || "").trim() : "";
    const status = statusIdx > -1 ? String(row[statusIdx] || "Active").trim() : "Active";
    const scheduledDate = dateIdx > -1 && row[dateIdx] ? String(row[dateIdx]).trim() : "";

    if (!catId && !catName) continue;

    const isScheduledFuture = scheduledDate && new Date(scheduledDate) > now;
    
    // Customer filter: hide archived/hidden categories and unreleased scheduled ones unless includeHidden is true
    if (!includeHidden) {
      if (status === "Hidden" || status === "Archived") continue;
      if (isScheduledFuture) continue;
    }

    list.push({
      id: catId || ("CAT-" + i),
      name: catName,
      parentId: parentId || null,
      status: status,
      scheduledLiveDate: scheduledDate || null,
      isScheduled: Boolean(isScheduledFuture)
    });
  }

  // Build nested category tree
  const categoryMap = {};
  list.forEach(c => { categoryMap[c.id] = { ...c, subcategories: [] }; });

  const rootCategories = [];
  list.forEach(c => {
    if (c.parentId && categoryMap[c.parentId]) {
      categoryMap[c.parentId].subcategories.push(categoryMap[c.id]);
    } else {
      rootCategories.push(categoryMap[c.id]);
    }
  });

  return rootCategories;
}

function saveCategory(categoryData) {
  if (!categoryData || !categoryData.name) {
    return {status: "error", message: "Category name is required."};
  }

  const db = getMasterDB();
  let sheet = db.getSheetByName("Categories");
  if (!sheet) {
    sheet = db.insertSheet("Categories");
    sheet.appendRow(["CategoryID", "CategoryName", "ParentCategoryID", "Status", "Scheduled_Live_Date"]);
  }

  const catId = categoryData.id || ("CAT-" + Date.now());
  const catName = String(categoryData.name).trim();
  const parentId = categoryData.parentId ? String(categoryData.parentId).trim() : "";
  const status = categoryData.status || "Active"; // Active | Hidden | Archived
  const scheduledDate = categoryData.scheduledLiveDate || "";

  const data = sheet.getDataRange().getValues();
  let existingRow = -1;

  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === catId) {
      existingRow = i + 1;
      break;
    }
  }

  if (existingRow > 0) {
    sheet.getRange(existingRow, 2).setValue(catName);
    sheet.getRange(existingRow, 3).setValue(parentId);
    sheet.getRange(existingRow, 4).setValue(status);
    sheet.getRange(existingRow, 5).setValue(scheduledDate);
  } else {

function getProductsFromSheet(includeScheduled) {
  const data = getUniversalData("Products");
  const now = new Date();

  return data.map(obj => {
    // Process primary Image_URL
    if (obj.Image_URL) {
      let urls = String(obj.Image_URL).split(',');
      let processedUrls = urls.map(u => {
        let trimmed = u.trim();
        if (trimmed.includes("drive.google.com") && !trimmed.includes("uc?export=view")) {
          const fileId = trimmed.match(/[-\w]{25,}/);
          if (fileId) return `https://drive.google.com/uc?export=view&id=${fileId[0]}`;
        }
        return trimmed;
      });
      obj.Image_URL = processedUrls.join(',');
    }

    // Process unlimited Media_URLs array (photos + videos)
    if (obj.Media_URLs) {
      try {
        obj.mediaList = typeof obj.Media_URLs === "string" ? JSON.parse(obj.Media_URLs) : obj.Media_URLs;
      } catch(e) {
        obj.mediaList = [];
      }
    } else {
      obj.mediaList = [];
    }

    // COD availability (default true if unspecified)
    obj.codAvailable = obj.COD_Available !== undefined && obj.COD_Available !== null 
      ? (String(obj.COD_Available).toLowerCase() === "true" || obj.COD_Available === true)
      : true;

    // Auto-Stock Scheduling (Coming Soon)
    if (obj.Scheduled_Live_Date) {
      const schedTime = new Date(obj.Scheduled_Live_Date);
      obj.isComingSoon = schedTime > now;
      obj.liveDateTimestamp = schedTime.getTime();
    } else {
      obj.isComingSoon = false;
    }

    return obj;
  }).filter(product => {
    // If regular customer browsing, include products unless admin marked inactive or hidden
    if (!includeScheduled && product.Status === "Hidden") return false;
    return true;
  });
}

function saveProduct(productData) {
  if (!productData || !productData.name) {
    return {status: "error", message: "Product name is required."};
  }

  const db = getMasterDB();
  let sheet = db.getSheetByName("Products");
  if (!sheet) {
    sheet = db.insertSheet("Products");
    sheet.appendRow(["ID", "Name", "Category", "ParentCategory", "Price", "MRP", "Stock", "Image_URL", "Media_URLs", "Description", "Status", "COD_Available", "Scheduled_Live_Date"]);
  }

  const prodId = productData.id || ("PROD-" + Date.now());
  const serializedMedia = typeof productData.mediaList === "object" ? JSON.stringify(productData.mediaList || []) : (productData.Media_URLs || "[]");
  const codAvailable = productData.codAvailable !== false && productData.COD_Available !== false; // boolean
  const scheduledDate = productData.scheduledLiveDate || productData.Scheduled_Live_Date || "";

  const rowObject = {
    "ID": prodId,
    "Name": String(productData.name || "").trim(),
    "Category": String(productData.category || "General").trim(),
    "ParentCategory": String(productData.parentCategory || "").trim(),
    "Price": Number(productData.price || 0),
    "MRP": Number(productData.mrp || productData.price || 0),
    "Stock": Number(productData.stock || 0),
    "Image_URL": String(productData.imageUrl || productData.Image_URL || ""),
    "Media_URLs": serializedMedia,
    "Description": String(productData.description || ""),
    "Status": String(productData.status || "Active"),
    "COD_Available": codAvailable,
    "Scheduled_Live_Date": scheduledDate
  };

  const headers = ensureDynamicColumns(sheet, rowObject);
  const data = sheet.getDataRange().getValues();
  const idIdx = headers.findIndex(h => String(h).toLowerCase() === "id");

  let existingRow = -1;
  if (idIdx > -1) {
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][idIdx]).trim() === prodId) {
        existingRow = i + 1;
        break;
      }
    }
  }

  let rowValues = headers.map(h => {
    let val = rowObject[h];
    if (typeof val === "object" && val !== null) val = JSON.stringify(val);
    return val !== undefined ? val : "";
  });

  if (existingRow > 0) {
    sheet.getRange(existingRow, 1, 1, headers.length).setValues([rowValues]);
  } else {
    sheet.appendRow(rowValues);
  }

  SpreadsheetApp.flush();
  return {status: "success", productId: prodId, message: "Product saved successfully with full media and delivery controls."};
}

function deleteProduct(productId) {
  if (!productId) return {status: "error", message: "Product ID is required."};
  const db = getMasterDB();
  const productSheet = db.getSheetByName("Products");
  if (!productSheet) return {status: "error", message: "Products sheet not found."};
  
  const lastRow = productSheet.getLastRow();
  if (lastRow <= 1) return {status: "error", message: "No products exist."};

  const data = productSheet.getDataRange().getValues();
  const headers = data[0];
  const idIndex = headers.findIndex(h => String(h).toLowerCase() === "id");
  
  if (idIndex === -1) return {status: "error", message: "ID column not found in Products sheet."};
  
  let deleted = false;
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idIndex]).trim() === String(productId).trim()) {
      productSheet.deleteRow(i + 1);
      deleted = true;
      break;
    }
  }

  if (!deleted) return {status: "error", message: "Product ID '" + productId + "' not found."};

  // Clean up waitlist and reviews safely in reverse
  const waitlistSheet = db.getSheetByName("Waitlist");
  if (waitlistSheet && waitlistSheet.getLastRow() > 1) {
    const wlData = waitlistSheet.getDataRange().getValues();
    for (let j = wlData.length - 1; j > 0; j--) {
      if (String(wlData[j][0]).trim() === String(productId).trim()) {
        waitlistSheet.deleteRow(j + 1);
      }
    }
  }
  
  const reviewSheet = db.getSheetByName("Reviews");
  if (reviewSheet && reviewSheet.getLastRow() > 1) {
    const rvData = reviewSheet.getDataRange().getValues();
    for (let k = rvData.length - 1; k > 0; k--) {
      if (String(rvData[k][1]).trim() === String(productId).trim()) {
        reviewSheet.deleteRow(k + 1);
      }
    }
  }
  
  SpreadsheetApp.flush();
  return {status: "success", message: `Product '${productId}' and associated records deleted successfully.`};
}

/**
 * Enriches a raw order record from the database with parsed items,
 * computed status stage, timeline, and tracking details.
 */
function enrichOrderTrackingDetails(order) {
  if (!order) return null;

  // 1. Parse items safely
  let itemsList = [];
  if (order.Items) {
    if (typeof order.Items === "string") {
      try {
        itemsList = JSON.parse(order.Items);
      } catch (e) {
        itemsList = [{ name: String(order.Items), quantity: 1, price: Number(order.Amount || 0) }];
      }
    } else if (Array.isArray(order.Items)) {
      itemsList = order.Items;
    }
  }

  // 2. Normalized status
  const rawStatus = String(order.Status || "Pending").trim();
  const validStatuses = ["Pending", "Confirmed", "Processing", "Shipped", "Out for Delivery", "Delivered", "Cancelled"];
  const matchedStatus = validStatuses.find(s => s.toLowerCase() === rawStatus.toLowerCase()) || rawStatus;

  // 3. Computed Status Progression Stages (1 to 5)
  const statusLevels = {
    "pending": 1,
    "confirmed": 2,
    "processing": 3,
    "shipped": 4,
    "out for delivery": 5,
    "delivered": 6,
    "cancelled": -1
  };
  const currentLevel = statusLevels[matchedStatus.toLowerCase()] || 1;

  const orderDateStr = order.Date ? new Date(order.Date).toLocaleString('en-IN', { dateStyle: 'medium', timeStyle: 'short' }) : "Order Placed";
  const estimatedArrival = order.Estimated_Delivery || order.EstimatedDelivery || "Within 3 - 5 business days";

  // Build interactive timeline steps
  const timeline = [
    {
      step: 1,
      key: "placed",
      title: "Order Placed",
      description: "Order received and queued for verification",
      timestamp: orderDateStr,
      completed: currentLevel >= 1,
      current: currentLevel === 1
    },
    {
      step: 2,
      key: "confirmed",
      title: "Order Confirmed",
      description: "Payment & inventory verified",
      timestamp: currentLevel >= 2 ? "Confirmed" : "Pending",
      completed: currentLevel >= 2,
      current: currentLevel === 2
    },
    {
      step: 3,
      key: "processing",
      title: "Packed in Warehouse",
      description: "Products safely packaged and quality checked",
      timestamp: currentLevel >= 3 ? "Packed" : "Pending",
      completed: currentLevel >= 3,
      current: currentLevel === 3
    },
    {
      step: 4,
      key: "shipped",
      title: "Dispatched with Courier",
      description: order.Courier_Name ? `Handed over to ${order.Courier_Name}` : "In transit to fulfillment hub",
      timestamp: currentLevel >= 4 ? (order.Tracking_Number ? `AWB: ${order.Tracking_Number}` : "In Transit") : "Pending",
      completed: currentLevel >= 4,
      current: currentLevel === 4
    },
    {
      step: 5,
      key: "out_for_delivery",
      title: "Out for Delivery",
      description: "Package is on the local courier delivery van",
      timestamp: currentLevel >= 5 ? "Out for delivery" : "Pending",
      completed: currentLevel >= 5,
      current: currentLevel === 5
    },
    {
      step: 6,
      key: "delivered",
      title: "Delivered",
      description: "Package delivered to recipient",
      timestamp: currentLevel >= 6 ? "Delivered" : `Est. ${estimatedArrival}`,
      completed: currentLevel >= 6,
      current: currentLevel === 6
    }
  ];

  return {
    orderId: order.Order_ID || order.OrderID,
    email: order.Email,
    amount: Number(order.Amount || 0),
    paymentMethod: order.Razorpay_ID === "COD" ? "COD" : "RAZORPAY",
    razorpayId: order.Razorpay_ID || "",
    status: matchedStatus,
    date: order.Date || new Date().toISOString(),
    deliveryAddress: order.Delivery_Address || order.DeliveryAddress || order.address || "Profile Address",
    courierName: order.Courier_Name || order.CourierName || "",
    trackingNumber: order.Tracking_Number || order.TrackingNumber || "",
    estimatedDelivery: estimatedArrival,
    items: itemsList,
    timeline: timeline,
    isCancelled: matchedStatus.toLowerCase() === "cancelled"
  };
}

/**
 * Tracks an order by Order ID (e.g. ORD-172651234), Email, or Tracking / Razorpay ID.
 * Fixes all previous order tracking bugs.
 */
function trackOrder(query) {
  if (!query) {
    return { status: "error", message: "Order ID or Email address is required to track an order." };
  }

  const cleanQuery = String(query).trim().toLowerCase();
  const allOrders = getUniversalData("Orders");
  if (!allOrders || allOrders.length === 0) {
    return { status: "error", message: "No order records found in the store system." };
  }

  // Filter matching records
  const matched = allOrders.filter(row => {
    const rawId = String(row.Order_ID || row.OrderID || "").toLowerCase().trim();
    const cleanId = rawId.replace(/^ord-/, "");
    const qClean = cleanQuery.replace(/^ord-/, "");

    const idMatch = rawId === cleanQuery || (cleanId.length > 3 && cleanId === qClean);
    const emailMatch = String(row.Email || "").toLowerCase().trim() === cleanQuery;
    const rzpMatch = String(row.Razorpay_ID || row.RazorpayID || "").toLowerCase().trim() === cleanQuery;
    const awbMatch = String(row.Tracking_Number || row.TrackingNumber || "").toLowerCase().trim() === cleanQuery;

    return idMatch || emailMatch || rzpMatch || awbMatch;
  });

  if (matched.length === 0) {
    return {
      status: "error",
      message: `Could not locate shipment for reference "${query}". Please verify your Order ID (format: ORD-...) or registered email address.`
    };
  }

  const enrichedOrders = matched.map(o => enrichOrderTrackingDetails(o));

  return {
    status: "success",
    orders: enrichedOrders,
    order: enrichedOrders[0]
  };
}

function getOrderHistory(email) {
  if (!email) return [];
  const result = trackOrder(email);
  return result.status === "success" ? result.orders : [];
}

function processOrder(orderData) {
  if (!orderData || !orderData.email) {
    return {status: "error", message: "Customer email is required to process order."};
  }
  const db = getMasterDB();
  
  // Verification for COD: check if COD was requested but disabled for any item
  if (orderData.razorpayId === "COD") {
    if (orderData.items) {
      let itemsList = [];
      try {
        itemsList = typeof orderData.items === "string" ? JSON.parse(orderData.items) : orderData.items;
      } catch(e) {}

      // Cross check against products
      const allProducts = getProductsFromSheet(true);
      for (let cartItem of itemsList) {
        let prod = allProducts.find(p => String(p.ID) === String(cartItem.id));
        if (prod && prod.codAvailable === false) {
          return {
            status: "error", 
            message: `Cash on Delivery is not available for '${prod.Name}'. Please choose Online Payment via Razorpay.`
          };
        }
      }
    }
  }

  // Razorpay Online Payment Verification
  if (orderData.razorpayId && String(orderData.razorpayId).trim() !== "COD") {
    try {
      const rzpKeyId = PropertiesService.getScriptProperties().getProperty("RAZORPAY_KEY_ID") || (typeof RAZORPAY_LIVE_KEY !== "undefined" ? RAZORPAY_LIVE_KEY : "rzp_live_SVstISgPrcivjP");
      const rzpKeySecret = PropertiesService.getScriptProperties().getProperty("RAZORPAY_KEY_SECRET") || "SFRSwjiLdIxkgO02f6rOF2QP";
      
      const auth = Utilities.base64Encode(rzpKeyId + ":" + rzpKeySecret);
      const rzpOptions = {
        "method": "get",
        "headers": { "Authorization": "Basic " + auth },
        "muteHttpExceptions": true
      };
      
      const response = UrlFetchApp.fetch("https://api.razorpay.com/v1/payments/" + orderData.razorpayId, rzpOptions);
      const resCode = response.getResponseCode();
      const rzpData = JSON.parse(response.getContentText());
      
      if (resCode !== 200 || (rzpData.status !== "captured" && rzpData.status !== "authorized")) {
        return {status: "error", message: "Razorpay payment verification failed: " + (rzpData.error ? rzpData.error.description : "Payment not captured.")};
      }
    } catch(e) {
      console.warn("Payment verification exception:", e.message);
    }
  }

  const orderId = "ORD-" + Date.now();
  const orderDate = new Date();
  
  let orderSheet = db.getSheetByName("Orders");
  if (!orderSheet) {
    orderSheet = db.insertSheet("Orders");
    orderSheet.appendRow(["Order_ID", "Email", "Items", "Amount", "Razorpay_ID", "Status", "Date", "Delivery_Address", "Courier_Name", "Tracking_Number", "Estimated_Delivery"]);
  }

  const serializedItems = typeof orderData.items === "object" ? JSON.stringify(orderData.items) : String(orderData.items || "[]");
  const deliveryAddr = orderData.Delivery_Address || orderData.deliveryAddress || orderData.address || "Default Profile Address";

  let oHeaders = ensureDynamicColumns(orderSheet, orderData);
  let newRow = new Array(oHeaders.length).fill("");

  let idIdx = oHeaders.indexOf("Order_ID");
  let emailIdx = oHeaders.indexOf("Email");
  let itemsIdx = oHeaders.indexOf("Items");
  let amountIdx = oHeaders.indexOf("Amount");
  let rzpIdx = oHeaders.indexOf("Razorpay_ID");
  let statusIdx = oHeaders.indexOf("Status");
  let dateIdx = oHeaders.indexOf("Date");
  let addrIdx = oHeaders.indexOf("Delivery_Address");
  let courierIdx = oHeaders.indexOf("Courier_Name");
  let trackIdx = oHeaders.indexOf("Tracking_Number");
  let estIdx = oHeaders.indexOf("Estimated_Delivery");

  if (idIdx > -1) newRow[idIdx] = orderId;
  if (emailIdx > -1) newRow[emailIdx] = String(orderData.email).toLowerCase().trim();
  if (itemsIdx > -1) newRow[itemsIdx] = serializedItems;
  if (amountIdx > -1) newRow[amountIdx] = Number(orderData.amount || 0);
  if (rzpIdx > -1) newRow[rzpIdx] = orderData.razorpayId || "COD";
  if (statusIdx > -1) newRow[statusIdx] = "Pending";
  if (dateIdx > -1) newRow[dateIdx] = orderDate.toISOString();
  if (addrIdx > -1) newRow[addrIdx] = deliveryAddr;
  if (courierIdx > -1) newRow[courierIdx] = "";
  if (trackIdx > -1) newRow[trackIdx] = "";
  if (estIdx > -1) newRow[estIdx] = "Within 3 - 5 business days";
  
  orderSheet.appendRow(newRow);
  syncCart(orderData.email, "[]");

  // Sync to visual customer vault
  const usersSheet = db.getSheetByName("Users");
  if (usersSheet) {
    const rowIndex = findRowIndexByEmailCaseInsensitive(orderData.email, usersSheet);
    if (rowIndex > 0) {
      let uHeaders = usersSheet.getRange(1, 1, 1, usersSheet.getLastColumn()).getValues()[0];
      let vaultIdx = uHeaders.findIndex(h => String(h).toLowerCase() === "vaultid");
      if (vaultIdx > -1) {
        let vaultId = usersSheet.getRange(rowIndex, vaultIdx + 1).getValue();
        if (vaultId) {
          try { 
            SpreadsheetApp.openById(vaultId).getSheetByName("Profile_And_Orders").appendRow([orderId, serializedItems, orderData.amount, orderDate.toLocaleDateString(), "Pending"]); 
          } catch(e) {} 
        }
      }
    }
  }

  SpreadsheetApp.flush();

  try {
    sendOrderReceivedEmail(orderData, orderId);
  } catch(e) {}

  const rawOrder = {
    Order_ID: orderId,
    Email: orderData.email,
    Items: serializedItems,
    Amount: orderData.amount,
    Razorpay_ID: orderData.razorpayId || "COD",
    Status: "Pending",
    Date: orderDate.toISOString(),
    Delivery_Address: deliveryAddr,
    Courier_Name: "",
    Tracking_Number: "",
    Estimated_Delivery: "Within 3 - 5 business days"
  };
  const enriched = enrichOrderTrackingDetails(rawOrder);

  return {
    status: "success", 
    orderId: orderId, 
    order: enriched,
    message: "Order placed successfully."
  };
}

  const sheet = getMasterDB().getSheetByName("Users");

  const rowIndex = findRowIndexByEmailCaseInsensitive(email, sheet);
  if (rowIndex > 0) {
    let headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    let lastSeenIdx = headers.findIndex(h => String(h).toLowerCase() === "lastseen");
    
    if (lastSeenIdx === -1) {
      sheet.getRange(1, headers.length + 1).setValue("LastSeen");
      lastSeenIdx = headers.length;
    }
    sheet.getRange(rowIndex, lastSeenIdx + 1).setValue(details || new Date().toISOString());
    SpreadsheetApp.flush();
  }
  return {status: "success"};
}

function updateOrderStatusAndAlert(orderId, newStatus, courierName, trackingNumber, estDelivery) {
  if (!orderId || !newStatus) return {status: "error", message: "Order ID and new Status are required."};
  const sheet = getMasterDB().getSheetByName("Orders");
  if (!sheet) return {status: "error", message: "Orders sheet not found."};

  // Ensure Courier_Name, Tracking_Number, Estimated_Delivery columns exist
  let headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const requiredCols = ["Courier_Name", "Tracking_Number", "Estimated_Delivery"];
  requiredCols.forEach(col => {
    const found = headers.some(h => String(h).toLowerCase().replace(/_/g, "") === col.toLowerCase().replace(/_/g, ""));
    if (!found) {
      sheet.getRange(1, sheet.getLastColumn() + 1).setValue(col);
      headers.push(col);
    }
  });

  const data = sheet.getDataRange().getValues();
  const orderIdIdx = headers.findIndex(h => String(h).toLowerCase().replace(/_/g, "") === "orderid");
  const statusIdx = headers.findIndex(h => String(h).toLowerCase() === "status");
  const emailIdx = headers.findIndex(h => String(h).toLowerCase() === "email");
  const courierIdx = headers.findIndex(h => String(h).toLowerCase().replace(/_/g, "") === "couriername");
  const trackingIdx = headers.findIndex(h => String(h).toLowerCase().replace(/_/g, "") === "trackingnumber");
  const estIdx = headers.findIndex(h => String(h).toLowerCase().replace(/_/g, "") === "estimateddelivery");

  for (let i = 1; i < data.length; i++) {
    const rowOrderId = String(data[i][orderIdIdx]).trim();
    if (rowOrderId === String(orderId).trim() || rowOrderId.replace(/^ord-/, "") === String(orderId).trim().replace(/^ord-/, "")) {
      // 1. Update Status
      if (statusIdx > -1) sheet.getRange(i + 1, statusIdx + 1).setValue(newStatus);

      // 2. Update Courier info if provided
      if (courierName && courierIdx > -1) sheet.getRange(i + 1, courierIdx + 1).setValue(courierName);
      if (trackingNumber && trackingIdx > -1) sheet.getRange(i + 1, trackingIdx + 1).setValue(trackingNumber);
      if (estDelivery && estIdx > -1) sheet.getRange(i + 1, estIdx + 1).setValue(estDelivery);

      SpreadsheetApp.flush(); 

      const email = emailIdx > -1 ? data[i][emailIdx] : "";
      const effectiveCourier = courierName || (courierIdx > -1 ? data[i][courierIdx] : "");
      const effectiveAwb = trackingNumber || (trackingIdx > -1 ? data[i][trackingIdx] : "");
      const effectiveEst = estDelivery || (estIdx > -1 ? data[i][estIdx] : "");

      // Automated email triggers with SVG icons
      if (email) {
        if (String(newStatus).toLowerCase() === "shipped") {
          try {
            sendOrderShippedEmail(email, rowOrderId, effectiveCourier, effectiveAwb, effectiveEst);
          } catch(e) {}
        } else if (String(newStatus).toLowerCase() === "delivered") {
          try {
            sendOrderDeliveredEmail(email, rowOrderId);
          } catch(e) {}
        }
      }

      return {
        status: "success", 
        message: `Order #${rowOrderId} updated to '${newStatus}' with tracking info.`,
        orderId: rowOrderId,
        newStatus: newStatus
      };
    }
  }
  return {status: "error", message: `Order #${orderId} not found.`};
}

function sendBulkPromo(title, message, bannerUrl) {
  if (!title || !message) return {status: "error", message: "Title and message are required."};
  const usersSheet = getMasterDB().getSheetByName("Users");
  if (!usersSheet || usersSheet.getLastRow() <= 1) return {status: "error", message: "No users found."};

  const data = usersSheet.getDataRange().getValues();
  const emailColIdx = data[0].findIndex(h => String(h).toLowerCase() === "email");
  const idx = emailColIdx > -1 ? emailColIdx : 1;

  let emailList = [];
  for (let i = 1; i < data.length; i++) {
    let val = data[i][idx];
    if (val && String(val).includes("@")) emailList.push(String(val).trim());
  }

  emailList = Array.from(new Set(emailList));
  const BATCH_SIZE = 40;
  for (let i = 0; i < emailList.length; i += BATCH_SIZE) {
    const chunk = emailList.slice(i, i + BATCH_SIZE).join(",");
    try {
      sendBulkPromoEmail(chunk, title, message, bannerUrl);
    } catch(e) {}
  }
  return {status: "success", message: `Promo sent to ${emailList.length} customers.`};
}

function getBannersFromSheet(includeInactive) {
  const db = getMasterDB();
  let sheet = db.getSheetByName("Banners");
  if (!sheet || sheet.getLastRow() <= 1) {
    return [];
  }

  const data = sheet.getDataRange().getValues();
  const rawHeaders = data[0].map(h => String(h).trim());
  const cleanHeaders = rawHeaders.map(h => h.toLowerCase().replace(/[^a-z0-9]/g, ""));
  
  const findCol = function(possibleKeys) {
    for (var k = 0; k < possibleKeys.length; k++) {
      var norm = possibleKeys[k].toLowerCase().replace(/[^a-z0-9]/g, "");
      var idx = cleanHeaders.indexOf(norm);
      if (idx > -1) return idx;
    }
    return -1;
  };

  const idIdx = findCol(["bannerid", "id", "banner_id"]);
  const titleIdx = findCol(["title", "name", "bannertitle", "heading"]);
  const subIdx = findCol(["subtitle", "sub_title", "description", "caption"]);
  const imgIdx = findCol(["imageurl", "image_url", "image", "photo", "bannerimage", "mediaurl", "url"]);
  const linkIdx = findCol(["linkurl", "link_url", "link", "targeturl", "target_url"]);
  const catIdx = findCol(["targetcategory", "category", "target_category"]);
  const badgeIdx = findCol(["badgetext", "badge", "badge_text", "tag"]);
  const btnIdx = findCol(["buttontext", "button", "button_text", "cta"]);
  const seqIdx = findCol(["sequence", "order", "seq", "priority", "index"]);
  const statusIdx = findCol(["status", "active"]);
  const dateIdx = findCol(["createdat", "date", "created_at"]);

  const banners = [];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const id = String(row[idIdx > -1 ? idIdx : 0] || "").trim();
    if (!id) continue;

    const status = statusIdx > -1 ? String(row[statusIdx] || "Active").trim() : "Active";
    if (!includeInactive && status.toLowerCase() !== "active") continue;

    banners.push({
      id: id,
      title: titleIdx > -1 ? String(row[titleIdx] || "") : "",
      subtitle: subIdx > -1 ? String(row[subIdx] || "") : "",
      imageUrl: imgIdx > -1 ? String(row[imgIdx] || "") : "",
      linkUrl: linkIdx > -1 ? String(row[linkIdx] || "") : "",
      targetCategory: catIdx > -1 ? String(row[catIdx] || "") : "",
      badgeText: badgeIdx > -1 ? String(row[badgeIdx] || "") : "",
      buttonText: btnIdx > -1 ? String(row[btnIdx] || "Shop Collection") : "Shop Collection",
      sequence: seqIdx > -1 ? Number(row[seqIdx] || i) : i,
      status: status,
      createdAt: dateIdx > -1 ? String(row[dateIdx] || "") : ""
    });
  }

  // Sort by sequence ascending
  banners.sort((a, b) => a.sequence - b.sequence);
  return banners;
}

    </div>

    <div style="background-color:#f8fafc; border-left:4px solid #065f46; padding:16px 20px; margin-bottom:24px; font-size:14px; line-height:1.7; color:#1e293b; white-space:pre-line;">
      ${replyText}
    </div>

    <p style="font-size:13px; color:#64748b;">
      Warm regards,<br>
      <strong>Aarambh Naturals Care Team</strong><br>
      <span style="font-size:11px; color:#94a3b8;">support@aarambhntpl.com</span>
    </p>
  `;

  try {
    MailApp.sendEmail({
      to: email,
      subject: "Response from Aarambh Naturals Care Team",
      name: "Aarambh Naturals Support",
      htmlBody: buildEmailHtml(content, "New message from Aarambh Naturals Care Team.")
    });

  } catch(e) {
    return {status: "error", message: e.message};
  }
}

/**
 * Sends bulk promotional announcement with zero emojis and responsive layout.
 */
function sendBulkPromoEmail(emailsBCC, title, message, bannerUrl) {
  if (!emailsBCC) return;
  const bccString = Array.isArray(emailsBCC) ? emailsBCC.join(",") : String(emailsBCC);
  if (!bccString.trim()) return;

  const bannerHtml = bannerUrl ? `
    <div style="margin-bottom:20px; text-align:center;">
      <img src="${bannerUrl}" alt="${title}" style="width:100%; max-width:540px; height:auto; border-radius:6px; display:block; margin:0 auto; border:1px solid #e2e8f0;">
    </div>
  ` : "";

  const content = `
    ${bannerHtml}
    <h2 style="margin:0 0 12px; font-size:22px; font-weight:700; color:#0f172a;">${title}</h2>
    <p style="font-size:14px; color:#475569; line-height:1.7; margin-bottom:24px; white-space:pre-line;">
      ${message}
    </p>
    <div style="text-align:center; margin-top:28px;">
      <a href="https://aarambhntpl.com" style="display:inline-block; padding:12px 30px; background-color:#065f46; color:#ffffff; text-decoration:none; font-weight:700; font-size:13px; border-radius:6px;">
        Shop the Collection
      </a>
    </div>
  `;

  let senderEmail = "";
  try {
    senderEmail = Session.getActiveUser().getEmail();
  } catch(e) {}

  try {
    MailApp.sendEmail({
      to: senderEmail || "notifications@aarambhntpl.com",
      bcc: bccString,
      subject: `Special Update: ${title}`,
      name: "Aarambh Naturals",
      htmlBody: buildEmailHtml(content, title)
    });
  } catch(e) {
    console.warn("sendBulkPromoEmail error:", e.message);
  }
}
